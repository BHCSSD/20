## Step 1: Map Layout & Player Variables

```javascript
let layout = {
  trees: [[2, 2], [2, 3], [5, 8], [6, 8]],
  mountains: [[0, 0], [1, 0], [0, 1]],
  roads: [[4, 0], [4, 1], [4, 2], [4, 3], [5, 3], [6, 3]]
};

let cols = 10;
let rows = 10;
let cellSize;

// Smooth-moving player sprite properties
let sprite = {
  cx: 160, // Starting X pixel position (col 4 * 40)
  cy: 0,   // Starting Y pixel position (row 0 * 40)
  size: 30,
  speed: 4
};
```

## Step 2: Main Loop & Setup

```javascript
function setup() {
  createCanvas(400, 400);
  cellSize = width / cols;
  textAlign(CENTER, CENTER);
}

function draw() {
  background(100, 200, 100);
  drawGridLines();

  // Draw environmental tiles
  drawFeature(layout.roads, color(150, 150, 150), "🛣️");
  drawFeature(layout.trees, color(34, 139, 34), "🌲");
  drawFeature(layout.mountains, color(112, 128, 144), "⛰️");

  // Predict movement and check against impassable obstacle lists
  movement([layout.trees, layout.mountains]);

  drawSprite();
}

function drawSprite() {
  fill("limegreen");
  stroke(0);
  rect(sprite.cx, sprite.cy, sprite.size, sprite.size);
}
```

## Step 3: Predictive Movement Logic

Predict where the player wants to go first (`nextX`, `nextY`), test if that position causes a collision, and only update coordinates if safe.

```javascript
function movement(obstacleLists) {
  // 1. Predict Next Position
  let nextX = sprite.cx;
  let nextY = sprite.cy;

  // 2. Read Keyboard Inputs (WASD / Arrows)
  if (keyIsDown(65) || keyIsDown(LEFT_ARROW))  { nextX -= sprite.speed; }
  if (keyIsDown(68) || keyIsDown(RIGHT_ARROW)) { nextX += sprite.speed; }
  if (keyIsDown(87) || keyIsDown(UP_ARROW))    { nextY -= sprite.speed; }
  if (keyIsDown(83) || keyIsDown(DOWN_ARROW))  { nextY += sprite.speed; }

  // 3. Keep within canvas boundaries
  nextX = constrain(nextX, 0, width - sprite.size);
  nextY = constrain(nextY, 0, height - sprite.size);

  // 4. Test predicting coordinates against all obstacle categories
  let canMove = true;
  for (let j = 0; j < obstacleLists.length; j++) {
    if (checkCollisions(nextX, nextY, obstacleLists[j], cellSize)) {
      canMove = false;
      break; // Stop checking if any collision is found
    }
  }

  // 5. Update actual player position only if safe
  if (canMove) {
    sprite.cx = nextX;
    sprite.cy = nextY;
  }
}
```

## Step 4: Grid-Aware Collision Helper

Convert grid positions `[col, row]` to pixels and perform bounding box checks.

```javascript
function checkCollisions(px, py, gridArray, size) {
  for (let i = 0; i < gridArray.length; i++) {
    // Convert tile [col, row] grid coordinates to canvas pixels
    let obstacleX = gridArray[i][0] * size;
    let obstacleY = gridArray[i][1] * size;

    // Optimization: Calculate distance between centers to skip far objects
    let distanceX = abs((px + sprite.size / 2) - (obstacleX + size / 2));
    let distanceY = abs((py + sprite.size / 2) - (obstacleY + size / 2));

    if (distanceX > size * 2 || distanceY > size * 2) {
      continue;
    }

    // Axis-Aligned Bounding Box (AABB) overlap math
    if (px < obstacleX + size &&
        px + sprite.size > obstacleX &&
        py < obstacleY + size &&
        py + sprite.size > obstacleY) {
      return true; // Collision detected!
    }
  }
  return false; // Path is clear
}

function drawFeature(featureArray, col, emoji) {
  for (let i = 0; i < featureArray.length; i++) {
    let pixelX = featureArray[i][0] * cellSize;
    let pixelY = featureArray[i][1] * cellSize;
    
    fill(col);
    noStroke();
    rect(pixelX, pixelY, cellSize, cellSize);
    
    fill(255);
    textSize(cellSize * 0.5);
    text(emoji, pixelX + cellSize / 2, pixelY + cellSize / 2);
  }
}

function drawGridLines() {
  stroke(0, 40);
  for (let i = 0; i <= cols; i++) {
    line(i * cellSize, 0, i * cellSize, height);
    line(0, i * cellSize, width, i * cellSize);
  }
}
```
