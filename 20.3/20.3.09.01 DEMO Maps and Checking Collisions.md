```javascript
function drawSprite() {
  // Translate sprite's grid coordinates [col, row] into canvas pixel positions
  let pixelX = sprite.col * cellSize
  let pixelY = sprite.row * cellSize

  // 1. Draw tile background square
  fill("limegreen")
  stroke(0)
  strokeWeight(2)
  rect(pixelX, pixelY, cellSize, cellSize)

  // 2. Draw centered emoji inside the cell
  fill(255)
  noStroke()
  textSize(cellSize * 0.6)
  textAlign(CENTER, CENTER)
  text(sprite.img, pixelX + cellSize / 2, pixelY + cellSize / 2)
}

```

## Step 4: Grid Collision Helper

Instead of using dist, we are going to use grid collision checks the player's target `[targetCol, targetRow]` against obstacle grid coordinates.
* No need to change anything in this this. 

```javascript
function isObstacle(targetCol, targetRow, obstacleLists) {

  //outer loop, goes through each obstacle category array  ( [layout.trees, layout.mountains])
  for (let i = 0; i < obstacleLists.length; i++) {
    let currentList = obstacleLists[i]  
    
    //innerloop,  goes over each individual [col, row] inside the current obstacle
    for (let j = 0; j < currentList.length; j++) {
      let obsCol = currentList[j][0]
      let obsRow = currentList[j][1]

      // If predicted cell matches an obstacle cell, collision detected
      if (targetCol === obsCol && targetRow === obsRow) {
        return true
      }//end of if

    }//end of nested for
  }//end of outer for

  return false // Cell is clear
}//end of isObstacle

```
