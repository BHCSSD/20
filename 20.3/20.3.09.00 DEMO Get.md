## Reading Colours with `get()` 
The `get()` function allows you to pick up the colour of any pixel on the canvas.
* `get(x, y)` returns an array containing 4 RGBA values: `[Red, Green, Blue, Alpha]`.
  * * Values range from `0` to `255`.

---

## Starter Code

Complete **Step 1**, **Step 2**, and **Step 3** in the sketch below.

```javascript
function setup() {
  createCanvas(400, 400);
}

function draw() {
  background(255);
  drawBackground();
  
  // STEP 1: Grab the color array at mouse location
  let sampledColor = get(mouseX, mouseY);
  
  // STEP 2: get individual RGB channels
  let r = sampledColor[0];
  let g = sampledColor[1];
  let b = sampledColor[2];

  // STEP 3: Display values in the inspection box
  displayBox(r, g, b);

  drawMouse();
}

function drawMouse() {
  push();
  stroke(255);
  strokeWeight(2);
  noFill();
  ellipse(mouseX, mouseY, 20, 20);
  pop();
}

function displayBox(r, g, b) {
  push();
  fill(0, 180);
  rect(10, 10, 130, 75, 5);
  
  fill(255);
  textSize(14);
  textAlign(LEFT, CENTER);
  text(`Red:   ${r}`, 20, 30);
  text(`Green: ${g}`, 20, 50);
  text(`Blue:  ${b}`, 20, 70);
  pop();
}


function drawBackground() {
  noStroke();
  
  // Red Quadrant
  fill(255, 100, 100);
  rect(0, 0, 200, 200);

  // Green Quadrant
  fill(100, 255, 100);
  rect(200, 0, 200, 200);

  // Blue Quadrant
  fill(100, 100, 255);
  rect(0, 200, 200, 200);

  // Yellow Quadrant
  fill(255, 200, 50);
  rect(200, 200, 200, 200);
}
```
