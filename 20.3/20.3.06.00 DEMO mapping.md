## Step 1: Defining the Map Layout

Instead of hardcoding raw pixel values, we use a grid object. Each map feature contains an array of `[column, row]` coordinate pairs.

```javascript
let layout = {
  trees: [[2, 2], [2, 3], [5, 8], [6, 8]],
  mountains: [[0, 0], [1, 0], [0, 1]],
  roads: [[4, 0], [4, 1], [4, 2], [4, 3], [5, 3], [6, 3]]
};

let cols = 10
let rows = 10
let cellSize; //calculated dynamically in setup()
```

---

## Step 2: Canvas & Grid Initialization

Calculate `cellSize` dynamically by dividing canvas width by total columns. This makes sure the map auto scales

```javascript
function setup() {
  createCanvas(400, 400)
  cellSize = width / cols; // 400/10 = 40 pixels per grid cell
  textAlign(CENTER, CENTER)
}

function draw() {
  background(100, 200, 100)
  
  drawGridLines()

  // Draw features using layout object coordinates
  drawFeature(layout.roads, color(150, 150, 150), "🛣️")
  drawFeature(layout.mountains, color(112, 128, 144), "⛰️")
  drawFeature(layout.trees, color(34, 139, 34), "🌲")
}
```

## Step 3: Drawing Helper Functions

Complete the helper functions below to map grid positions to actual canvas pixels.

```javascript
function drawFeature(featureArray, col, emoji) {
  for (let i = 0; i < featureArray.length; i++) {

    // 1. Extract grid coordinate pair [col, row]
    let gridX = featureArray[i][0];
    let gridY = featureArray[i][1];
    
    // 2. Convert grid coordinates to canvas pixel positions
    let pixelX = gridX * cellSize;
    let pixelY = gridY * cellSize;
    
    // 3. Draw tile background
    fill(col);
    noStroke();
    rect(pixelX, pixelY, cellSize, cellSize);
    
    // 4. Draw centered tile icon
    fill(255);
    textSize(cellSize * 0.5);
    text(emoji, pixelX + cellSize / 2, pixelY + cellSize / 2);
  }
}

function drawGridLines() {
  stroke(0, 40);
  for (let i = 0; i <= cols; i++) {
    line(i * cellSize, 0, i * cellSize, height);
    line(0, i * cellSize, width, i * cellSize);
  }
}
```
