function drawBigStar(cx, cy, starSize, col) {
  push()
  fill(col)
  stroke(0)
  strokeWeight(1)
  beginShape()
  for (let i = 0; i < 10; i++) {
    let angle = (i * 36) - 90; // 360 degrees / 10 points, offset by -90 to point straight up
    let radius = (i % 2 === 0) ? starSize : starSize / 2;
    let x = cx + cos(angle) * radius;
    let y = cy + sin(angle) * radius;
    vertex(x, y)
  }
  endShape(CLOSE)
  pop()
}
