## 2. View

The *View* handles all rendering and layout. It reads state variables to display dynamic text.

```javascript
// View: Handles display

function setup() {
  createCanvas(800, 600);
}

function draw() {
  background(200, 0, 80);
  
  header();
  displayResultsBox();
}

function header() {
  // Static header box
  fill(255);
  strokeWeight(3);
  stroke(50, 50, 100);
  rect(20, 20, width - 40, 180);

  fill(0);
  noStroke();
  textSize(24);
  text("Welcome to the new band name analyzer.", 50, 70);

  textSize(15);
  text("This website will judge how much money your band name will generate.", 50, 110);
  text("Press 'b' to enter your band name.", 50, 140);
}

function displayResultsBox() {
  // Reusable message container box
  fill(255);
  strokeWeight(3);
  stroke(50, 50, 100);
  rect(20, 220, width - 40, 320);

  // Render current status messages from Model state variables
  fill(0);
  noStroke();
  textSize(16);

  if (searchStatus !== "") {
    text(searchStatus, 50, 270);
  }
  if (resultStatus !== "") {
    text(resultStatus, 50, 320);
  }
  if (scoreStatus !== "") {
    text(scoreStatus, 50, 370);
  }
}
```
